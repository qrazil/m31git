# The command line of 1git, one case per option: the line in our grammar, the
# argv of git's grammar it is handed to the command function as (`=`), and the
# spellings git has for it that `1git -migrate` rewrites
# into the line above (`<`). tests/test_cli.sh runs all three ways and checks
# that every option of every command's --help is in here.
#
#   -log --max 5          what the user types
#   = log --max-count=5   what the command function receives
#   < git log -n 5        a git spelling, rewritten to the first line
#
# Words are shell-quoted.

= git status
-status --format short --branch
= git status --short --branch
< git status -sb
-status --format porcelain2 --nul --branch
= git status --porcelain=v2 --branch -z
< git status --porcelain=v2 -z -b

= git status --porcelain
< git status --porcelain=v1
-status --format long
= git status --long
-status --untracked all
= git status -uall
< git status -u
< git status --untracked-files=all
-status --untracked no --ignored matching
= git status -uno --ignored=matching
-status --untracked normal --ignored traditional
= git status -unormal --ignored
-status --ignored traditional
= git status --ignored
< git status --ignored
-status --renames off --stash
= git status --no-renames --show-stash
-status --renames on
= git status --renames
< git status -M
-status src docs
= git status -- src docs
