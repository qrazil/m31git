# The command line of 1git, one case per option: the line in our grammar, the
# argv of git's grammar it is handed to the command function as (`=`), and the
# spellings git has for it that `1git -migrate` rewrites
# into the line above (`<`). tests/test_cli.sh runs all three ways and checks
# that every option of every command's --help is in here.
#
#   -log --max 5          what the user types
#   = log --max-count=5   what the command function receives
#   < git log -n 5        a git spelling, rewritten to the first line
#
# Words are shell-quoted.

= git hash-object -w a.txt

< git commit-tree -F msg.txt main
-ls-files
= git ls-files
< git ls-files

-ls-files --others --exclude-standard

= git ls-files -z

# --- checkout (git's own: a branch or a file is -switch or -restore, the rest is this)
-checkout main
= git checkout main
-checkout main --path a.txt
= git checkout main -- a.txt
-checkout --create topic
= git checkout -b topic
-checkout --reset-create topic main
= git checkout -B topic main
-checkout main --detach --force --ours --theirs --track --patch --quiet
= git checkout --detach -f --ours --theirs --track -p -q main
-checkout --ours --path a.txt
= git checkout --ours -- a.txt
